## 2. 动作与能力点

```ts
export type CompetencyCode =
  | `K${1 | 2 | 3 | 4 | 5}`   // 知识
  | `S${1 | 2 | 3 | 4 | 5}`   // 技能
  | `A${1 | 2 | 3}`;          // 素养/安全

export interface Action {
  type: ActionType;
  choiceId: string;                                  // 定义文件里的唯一 id
  params?: Record<string, number | string | boolean>; // 体积/浓度/滴数/时间等
  competency: CompetencyCode;                        // 该动作判定哪个能力点
}
```

## 3. 实验状态

```ts
export interface EntityState {
  id: string;                                  // burette-1 / flask-1 / tube-1
  kind: "burette" | "flask" | "beaker" | "cylinder" | "pipette"
      | "volumetric" | "tube" | "balance" | "wash" | "stand" | "lamp";
  liquid?: { substance: string; level: number; color: string; opacity: number };
  precipitate?: { substance: string; amount: number; color: string };
  gas?: { substance: string; amount: number };
  bubbles?: boolean;
  reading?: number;                            // mL / g / pH
  note?: string;
}

export interface ExperimentState {
  expId: string;
  stepIndex: number;                           // 0-based
  attemptOfStep: number;                       // 本步第几次尝试（驱动提示阶梯）
  entities: Record<string, EntityState>;
  observations: Observation[];
  violations: ViolationRecord[];
  incidents: IncidentRecord[];
  done: boolean;
}

export interface Observation { stepIndex: number; text: string; value?: number }
export interface ViolationRecord { stepIndex: number; ruleId: string; deduct: number; competency: CompetencyCode }
export interface IncidentRecord { stepIndex: number; incidentId: string; level: IncidentLevel; terminate: boolean }
```

## 4. 事件（引擎输出，UI 只消费）

```ts
export type Event =
  | { kind: "invalid"; reason: string }                                   // 跳步/不合法，不改状态
  | { kind: "feedback"; severity: "info" | "warn" | "danger"; title: string; why?: string; correct?: string; refs?: string[] }
  | { kind: "hint"; level: HintLevel; text: string }
  | { kind: "stepComplete"; stepIndex: number }
  | { kind: "violation"; ruleId: string; deduct: number; competency: CompetencyCode }
  | { kind: "incident"; incidentId: string; level: IncidentLevel; fx: FxKey; terminate: boolean }
  | { kind: "observation"; text: string; value?: number }
  | { kind: "stateChanged"; patch: Partial<ExperimentState> };
```

引擎签名固定：

```ts
export function apply(state: ExperimentState, action: Action): { state: ExperimentState; events: Event[] };
```

纯函数，无 IO、无 React、无随机数。

## 5. 事故（安全层）

```ts
export type IncidentLevel = "L1" | "L2" | "L3" | "L4";   // 偏差 / 数据作废 / 事故 / 危险事故
export type FxKey =
  | "boil-over" | "explosion" | "toxic-gas" | "burn"
  | "glass-break" | "fire" | "heavy-metal" | "device-accident";

export interface Incident {
  id: string;                     // exp01-I1
  trigger: (state: ExperimentState, action: Action) => boolean;
  level: IncidentLevel;
  fx: FxKey;
  title: string;                  // 事故名，如"浓硫酸稀释放热爆沸"
  why: string;                    // 为什么错（原理，1~2 句）
  consequence: string;            // 真实后果
  correct: string;                // 正确做法（分步）
  refs: string[];                 // 规范条目 S001…
  deduct: number;                 // L1 2~3 / L2 5 / L3 8~10 / L4 15~20
  terminate: boolean;             // L4 一律 true
  competency: CompetencyCode;
}
```

**判分表（固定，不许各实验自定）**

| 级别 | 扣分 | 行为 |
|---|---|---|
| L1 | 2~3 | 提示，可继续 |
| L2 | 5 | 该步数据作废，必须重做 |
| L3 | 8~10 | 定格 + 事故卡 + 重做该步 |
| L4 | 15~20 | **终止本次实验** + 安全复盘 |

## 6. 提示阶梯

```ts
export type HintLevel = 0 | 1 | 2 | 3 | 4;
// 0 只给后果；1 指向性问题；2 概念提示；3 动作演示；4 元认知（连续 3 次同类错误）
```

规则：
- 第 1 次错 → H0；同一步第 2 次错 → H1；第 3 次 → H2；第 4 次 → H3；**连续 3 次同类错误**（跨步同 competency）→ H4。
- 用到 H2 及以上，每次 `independentDeduct += 1`（进操作分的"独立完成度"）。

## 7. 计分

```ts
opScore   = clamp(100 - Σviolation.deduct - Σincident.deduct - independentDeduct, 0, 100)
mastery(c)= clamp((correct(c) - 1.5 * wrong(c) - 3 * incidents(c)) / max(1, attempts(c)), 0, 1)
```

三项折算（对齐课程）：预习 10%（前测）+ 操作 40%（opScore）+ 报告 50%（数据/计算/误差/安全复盘）。
课程目标达成度：目标 1 = 前测+后测平均；目标 2 = opScore + mastery(S\*)；目标 3 = 1 − 事故率。

## 8. 时间线（落库形态，服务端重算分靠它）

```ts
export interface TimelineRow {
  runId: string;
  stepIndex: number;
  seq: number;              // 同一步内的序号，用于幂等
  actionType: ActionType;
  choiceId: string;
  competency: CompetencyCode;
  correct: boolean;
  deduct: number;
  incidentId?: string;

```

幂等键：`(runId, stepIndex, seq)` 唯一。

| POST | `/api/design/assess` | `{ planId }` | `{ assessment: string, offline: boolean }` |

全部 JSON，**不用 SSE**（扣子 90s 断连约束）。

## 10. `data-demo-id` 命名（演示脚本与 UI 的唯一契约）

```
demo-<page>-<role>-<n>
```

- page：`home` / `pregame` / `op` / `report` / `design` / `teacher`
- role：动作或元素名，kebab-case，语义化（`rinse-distilled` 而不是 `btn2`）
- 例：`demo-home-exp09`、`demo-op-step1-rinse-titrant`、`demo-op-titrate-half`、`demo-report-export`、`demo-design-validate`

规则：全局唯一；**新增交互元素必须同时给 id**；CI 遍历演示脚本里出现的每个 id，渲染树里找不到即红。

## 11. 视觉键 → 动画组件

`FxKey` 与 `components/fx/` 一一对应：`boil-over.tsx` `explosion.tsx` `toxic-gas.tsx` `burn.tsx` `glass-break.tsx` `fire.tsx` `heavy-metal.tsx` `device-accident.tsx`。
